```python
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        rows = len(grid)
        cols = len(grid[0])
        checked = [[0 for _ in range(cols)] for _ in range(rows)]
        def check_connected_islands(height, width):
            if not (0 <= height < rows and 0 <= width < cols):
                return
            if grid[height][width] == '0':
                return
            if checked[height][width]:
                return
            checked[height][width] = 1
            diffs = [(0, 1), (0, -1), (1, 0), (-1, 0)]
            for col_diff, row_diff in diffs:
                check_connected_islands(height + col_diff, width + row_diff)
        islands_num = 0
        for i in range(rows):
            for j in range(cols):
                if grid[i][j] == '0':
                    continue
                if checked[i][j]:
                    continue
                check_connected_islands(i, j)
                islands_num += 1
        return islands_num
```

```python
from collections import deque

class Solution:
    WATER = '0'
    LAND = '1'
    def numIslands(self, grid: List[List[str]]) -> int:
        width = len(grid[0])
        height = len(grid)
        visited = [[False]*width for _ in range(height)] 
        def find_connected_lands(row, column):
            lands = deque() 
            lands.append((row, column))
            while lands:
                current_row, current_column = lands.popleft()
                if not(0 <= current_row < height and 0 <= current_column < width):
                    continue
                if grid[current_row][current_column] == Solution.WATER:
                    continue
                if visited[current_row][current_column]:
                    continue
                visited[current_row][current_column] = True 
                diffs = [(0, 1), (0, -1), (1, 0), (-1, 0)] 
                for i, j in diffs:
                    lands.append((current_row + i, current_column + j))
            
        islands_num = 0
        for i in range(height):
            for j in range(width):
                if visited[i][j]:
                    continue
                if grid[i][j] == Solution.WATER:
                    continue
                islands_num += 1
                find_connected_lands(i, j)
        return islands_num
```

- UnionFindの自作

```python

class Node():
    def __init__(self):
        self.rank = 0
        self.parent = self
        

class UnionFind():    
    def make_set(self):
        return Node()
        
    def find_root(self, node):
        if node.parent == node:
            return node
        node.parent = self.find_root(node.parent)
        return node.parent
        
    def unite(self, node1, node2):
        root1 = self.find_root(node1)
        root2 = self.find_root(node2)
        if root1 == root2:
            return
        if root1.rank > root2.rank:
            root1, root2 = root2, root1
        root1.parent = root2
        root1.rank += 1

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        uf = UnionFind()
        nodes = [[None]*len(grid[0]) for _ in range(len(grid))]
        for h_i in range(len(grid)):
            for w_i in range(len(grid[0])):
                if grid[h_i][w_i] == '0':
                    continue
                nodes[h_i][w_i] = uf.make_set()
                if 0 <= h_i - 1 < len(grid) and nodes[h_i - 1][w_i]:
                    uf.unite(nodes[h_i - 1][w_i], nodes[h_i][w_i])
                if 0 <= w_i - 1 < len(grid[0]) and nodes[h_i][w_i - 1]:
                    uf.unite(nodes[h_i][w_i -1], nodes[h_i][w_i])
        root_nodes = set()
        for h_i in range(len(grid)):
            for w_i in range(len(grid[0])):
                if not nodes[h_i][w_i]:
                    continue
                root_nodes.add(uf.find_root(nodes[h_i][w_i]))
        return len(root_nodes)
```
