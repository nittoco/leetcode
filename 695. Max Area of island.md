```python
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        WATER = 0
        LAND = 1
        checked = [[False] * len(grid[0]) for _ in range(len(grid))]

        def calculate_area(start_row, start_col):

                return 0
            if grid[start_row][start_col] == WATER:
                return 0
            if checked[start_row][start_col]:
                return 0
            area = 1
            diffs = [(0, 1), (0, -1), (1, 0), (-1, 0)]
            checked[start_row][start_col] = True
            for row_d, col_d in diffs:
                area += calculate_area(start_row + row_d, start_col + col_d)
            return area

        max_area = 0
        for row_i in range(len(grid)):
            for col_i in range(len(grid[0])):
                if grid[row_i][col_i] == WATER:
                    continue
                if checked[row_i][col_i]:
                    continue
                max_area = max(max_area, calculate_area(row_i, col_i))
        return max_area
```

```python
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
        self.size = [1] * n

    def find_root(self, a):
        if self.parent[a] == a:
            return a
        self.parent[a] = self.find_root(self.parent[a])
        return self.parent[a]

    def unite(self, a, b):
        root_a = self.find_root(a)
        root_b = self.find_root(b)
        if root_a == root_b:
            return
        if self.rank[root_a] < self.rank[root_b]:
            self.parent[root_a] = root_b
            self.size[root_b] += self.size[root_a]
        else:
            self.parent[root_b] = root_a
            self.size[root_a] += self.size[root_b]
            if self.rank[root_a] == self.rank[root_b]:
                self.rank[root_a] += 1

    def get_size(self, a):
        return self.size[self.find_root(a)]

    def is_same(self, a, b):
        return self.find_root[a] == self.find_root[b]

class Solution:
    LAND = 1
    WATER = 0

    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        height = len(grid)
        width = len(grid[0])
        all_num = height * width
        index = lambda row, column: row * width + column
        uf = UnionFind(all_num)
        max_area = 0
        for r in range(height):
            for c in range(width):
                if grid[r][c] == self.WATER:
                    continue
                if r - 1 >= 0 and grid[r - 1][c] == self.LAND:
                    uf.unite(index(r - 1, c), index(r, c))
                if c - 1 >= 0 and grid[r][c - 1] == self.LAND:
                    uf.unite(index(r, c - 1), index(r, c))
                max_area = max(max_area, uf.get_size(index(r, c)))
        return max_area
```

```python
class Solution:
    LAND = 1
    WATER = 0

    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        def calculate_area(start_row, start_col):
            lands_stack = []
            lands_stack.append((start_row, start_col))
            inner_index = lambda row, col: 0 <= row < len(grid) and 0 <= col < len(grid[0])
            area = 0
            while lands_stack:
                row, col = lands_stack.pop()
                if not inner_index(row, col):
                    continue
                if grid[row][col] == self.WATER:
                    continue
                if checked[row][col]:
                    continue
                area += 1
                checked[row][col] = True
                diffs = [(0, 1), (0, -1), (1, 0), (-1, 0)]
                for row_d, col_d in diffs:
                    lands_stack.append((row + row_d, col + col_d))
            return area

        checked = [[False] * len(grid[0]) for _ in range(len(grid))]
        max_area = 0
        for r in range(len(grid)):
            for c in range(len(grid[0])):
                max_area = max(max_area, calculate_area(r, c))
        return max_area
```

```python
class Solution:
    LAND = 1
    WATER = 0
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        height = len(grid)
        width = len(grid[0])
        checked = [[False] * width for _ in range(height)]
        
        def is_inner_index(row, col):
            return 0 <= row < height and 0 <= col < width

        def calculate_area(row, col):
            if not is_inner_index(row, col):
                return 0
            if not grid[row][col] == self.LAND:
                return 0
            if checked[row][col]:
                return 0
            area = 1
            checked[row][col] = True
            diffs = [(0, 1), (0, -1), (1, 0), (-1, 0)]
            for row_diff, col_diff in diffs:
                area += calculate_area(row + row_diff, col + col_diff)
            return area

        max_area = 0
        for row in range(height):
            for col in range(width):
                max_area = max(max_area, calculate_area(row, col))
        return max_area
```
